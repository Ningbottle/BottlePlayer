Here's the core of the glass panel — the part with the least obvious mapping. Compact and self-contained.

Setup (once)
// Graphics.h — owns device + offscreen "backdrop" bitmap
struct Graphics {
    ComPtr<ID2D1Factory7>      d2dFactory;
    ComPtr<ID2D1Device6>       d2dDevice;
    ComPtr<ID2D1DeviceContext6> ctx;
    ComPtr<IDXGISwapChain1>    swap;
    ComPtr<ID2D1Bitmap1>       backbuffer;     // bound to swap chain
    ComPtr<ID2D1Bitmap1>       sceneBitmap;    // offscreen, full-res
    ComPtr<ID2D1Bitmap1>       blurredBitmap;  // offscreen, ¼-res

    ComPtr<ID2D1Effect>        gaussianBlur;
    ComPtr<ID2D1Effect>        saturation;
    ComPtr<ID2D1Effect>        shadow;

    bool blurDirty = true;  // recompute when scene changes
};
Creating the offscreen targets
void CreateOffscreens(Graphics& g, D2D1_SIZE_U size) {
    auto props = D2D1::BitmapProperties1(
        D2D1_BITMAP_OPTIONS_TARGET,
        D2D1::PixelFormat(DXGI_FORMAT_B8G8R8A8_UNORM, D2D1_ALPHA_MODE_PREMULTIPLIED));

    g.ctx->CreateBitmap(size, nullptr, 0, props, &g.sceneBitmap);

    D2D1_SIZE_U quarter{ size.width / 2, size.height / 2 };
    g.ctx->CreateBitmap(quarter, nullptr, 0, props, &g.blurredBitmap);

    g.ctx->CreateEffect(CLSID_D2D1GaussianBlur, &g.gaussianBlur);
    g.gaussianBlur->SetValue(D2D1_GAUSSIANBLUR_PROP_STANDARD_DEVIATION, 22.0f);
    g.gaussianBlur->SetValue(D2D1_GAUSSIANBLUR_PROP_BORDER_MODE,
                             D2D1_BORDER_MODE_HARD);

    g.ctx->CreateEffect(CLSID_D2D1ColorMatrix, &g.saturation);
    // boost saturation ~1.6× to feel "Apple-glass"
    D2D1_MATRIX_5X4_F sat = {
        1.36f, -0.18f, -0.18f, 0,
        -0.18f, 1.36f, -0.18f, 0,
        -0.18f, -0.18f, 1.36f, 0,
        0, 0, 0, 1,
        0, 0, 0, 0
    };
    g.saturation->SetValue(D2D1_COLORMATRIX_PROP_COLOR_MATRIX, sat);

    g.ctx->CreateEffect(CLSID_D2D1Shadow, &g.shadow);
    g.shadow->SetValue(D2D1_SHADOW_PROP_BLUR_STANDARD_DEVIATION, 18.0f);
    g.shadow->SetValue(D2D1_SHADOW_PROP_COLOR,
                       D2D1::Vector4F(0.16f, 0.11f, 0.05f, 0.35f));
}
Per-frame render order
void RenderFrame(Graphics& g, Scene& scene) {
    // 1) Draw EVERYTHING non-glass into sceneBitmap
    g.ctx->SetTarget(g.sceneBitmap.Get());
    g.ctx->BeginDraw();
    g.ctx->Clear(D2D1::ColorF(0.945f, 0.918f, 0.847f));  // paper
    DrawPaperTexture(g, scene);
    DrawSidebar(g, scene);
    DrawTopbar(g, scene);
    DrawCardGrid(g, scene);   // covers, halftoned
    g.ctx->EndDraw();

    // 2) Build the blurred backdrop (only when dirty)
    if (g.blurDirty) {
        g.ctx->SetTarget(g.blurredBitmap.Get());
        g.ctx->BeginDraw();
        g.gaussianBlur->SetInput(0, g.sceneBitmap.Get());
        g.saturation->SetInputEffect(0, g.gaussianBlur.Get());

        // Draw the effect graph scaled down to ¼ res
        g.ctx->SetTransform(D2D1::Matrix3x2F::Scale(0.5f, 0.5f));
        g.ctx->DrawImage(g.saturation.Get());
        g.ctx->SetTransform(D2D1::Matrix3x2F::Identity());
        g.ctx->EndDraw();
        g.blurDirty = false;
    }

    // 3) Composite to back buffer
    g.ctx->SetTarget(g.backbuffer.Get());
    g.ctx->BeginDraw();
    g.ctx->DrawBitmap(g.sceneBitmap.Get());               // base
    DrawGlassPanel(g, scene.drawer);                       // floating drawer
    DrawGlassPanel(g, scene.playerBar);                    // bottom bar
    DrawOverlayChrome(g, scene);                           // text, icons on top
    g.ctx->EndDraw();

    g.swap->Present(1, 0);
}
The glass panel itself
struct GlassPanel {
    D2D1_ROUNDED_RECT bounds;     // in DIPs
    D2D1_COLOR_F      tint;       // e.g. {0.97, 0.95, 0.88, 0.50}
};

void DrawGlassPanel(Graphics& g, const GlassPanel& p) {
    auto ctx = g.ctx.Get();

    // (a) drop shadow underneath
    g.shadow->SetInput(0, /* a bitmap mask of the rounded rect */ nullptr);
    // Simpler: just draw a soft fill below
    ComPtr<ID2D1SolidColorBrush> shadowBr;
    ctx->CreateSolidColorBrush(D2D1::ColorF(0.16f, 0.11f, 0.05f, 0.18f), &shadowBr);
    auto shRect = p.bounds;
    shRect.rect.top    += 12; shRect.rect.bottom += 24;
    shRect.rect.left   -= 4;  shRect.rect.right  += 4;
    ctx->FillRoundedRectangle(shRect, shadowBr.Get());

    // (b) clip to the panel's rounded rect
    ComPtr<ID2D1RoundedRectangleGeometry> geom;
    g.d2dFactory->CreateRoundedRectangleGeometry(p.bounds, &geom);

    ctx->PushLayer(
        D2D1::LayerParameters1(
            D2D1::InfiniteRect(), geom.Get(),
            D2D1_ANTIALIAS_MODE_PER_PRIMITIVE),
        nullptr);

    // (c) blit the cached blurred backdrop, upsampled
    D2D1_RECT_F src{
        p.bounds.rect.left   * 0.5f,
        p.bounds.rect.top    * 0.5f,
        p.bounds.rect.right  * 0.5f,
        p.bounds.rect.bottom * 0.5f
    };
    ctx->DrawBitmap(
        g.blurredBitmap.Get(),
        &p.bounds.rect, 1.0f,
        D2D1_INTERPOLATION_MODE_HIGH_QUALITY_CUBIC,
        &src);

    // (d) warm tint pass — gives the paper-glass feel
    ComPtr<ID2D1SolidColorBrush> tintBr;
    ctx->CreateSolidColorBrush(p.tint, &tintBr);
    ctx->FillRoundedRectangle(p.bounds, tintBr.Get());

    // (e) inner highlight (top edge)
    ComPtr<ID2D1LinearGradientBrush> highlight;
    {
        D2D1_GRADIENT_STOP stops[] = {
            { 0.0f, D2D1::ColorF(1, 0.99f, 0.95f, 0.70f) },
            { 0.15f, D2D1::ColorF(1, 0.99f, 0.95f, 0.00f) },
        };
        ComPtr<ID2D1GradientStopCollection> coll;
        ctx->CreateGradientStopCollection(stops, 2, &coll);
        ctx->CreateLinearGradientBrush(
            D2D1::LinearGradientBrushProperties(
                {p.bounds.rect.left,  p.bounds.rect.top},
                {p.bounds.rect.left,  p.bounds.rect.top + 60}),
            coll.Get(), &highlight);
    }
    ctx->FillRoundedRectangle(p.bounds, highlight.Get());

    ctx->PopLayer();

    // (f) crisp 1px hairline border (drawn OUTSIDE the clip)
    ComPtr<ID2D1SolidColorBrush> edgeBr;
    ctx->CreateSolidColorBrush(D2D1::ColorF(1, 0.99f, 0.95f, 0.85f), &edgeBr);
    ctx->DrawRoundedRectangle(p.bounds, edgeBr.Get(), 1.0f);
}
Things to remember
Mark blurDirty = true on scroll, drawer slide, window resize — anywhere the scene content under the glass changes. Don't re-blur every frame: it's the most expensive op in your pipeline.
The 0.5× scale in step 2 must match the * 0.5f in DrawGlassPanel step (c). If you change one, change both.
For the seek bar's ink-line nib, just draw a small FillEllipse with DrawLine underneath using a dashed ID2D1StrokeStyle1.
If the blurred edges look "soft" against the hairline border, draw the border with a StrokeStyle whose transformType = D2D1_STROKE_TRANSFORM_TYPE_HAIRLINE — guarantees 1 device pixel at any DPI/scale.
That's the whole trick. Once DrawGlassPanel works, the drawer and player bar are just two instances of it with different bounds and tints.
